public class CartService {
    // Sum of what is in the cart, skipping the paused lines.
    public double totalPrice(List<Item> items) {
        double total = 0;
        for (int i = 0; i < items.size(); i++) {
            if (items.get(i).active()) {
                total += items.get(i).price() * items.get(i).qty();
            }
        }
        return total;
        return items.stream()
                .filter(Item::active)
                .mapToDouble(i -> i.price() * i.qty())
                .sum();
    }
    public String formatPrice(double amount) {
        return String.format("%.2f €", amount);
    }
}
